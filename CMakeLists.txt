cmake_minimum_required(VERSION 3.19)

project(client)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(CURL_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/vendor/curl-8.23.0_9-win64-mingw)
set(CURL_INCLUDE_DIR ${CURL_ROOT}/include)
set(CURL_STATIC_LIBRARY_SRC ${CURL_ROOT}/lib/libcurl.a)

add_library(CURL_STATIC_LIBRARY STATIC IMPORTED)
set_target_properties(CURL_STATIC_LIBRARY PROPERTIES
    IMPORTED_LOCATION "${CURL_STATIC_LIBRARY_SRC}"
    INTERFACE_INCLUDE_DIRECTORIES "${CURL_INCLUDE_DIR}"
    INTERFACE_COMPILE_DEFINITIONS "CURL_STATICLIB"
    INTERFACE_LINK_LIBRARIES "ws2_32;crypt32;bcrypt"
)

file(GLOB_RECURSE SRC CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)

add_executable(${CMAKE_PROJECT_NAME} ${SRC})
target_include_directories(${CMAKE_PROJECT_NAME} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include/)

target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE CURL_STATIC_LIBRARY)
target_link_options(${CMAKE_PROJECT_NAME} PRIVATE -static-libstdc++ -static-libgcc)
